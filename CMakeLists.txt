cmake_minimum_required(VERSION 3.25)
project(ProjectionStudio VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PS_BUILD_EDITOR "Build the Qt desktop editor" ON)
option(PS_BUILD_TESTS "Build automated tests" ON)
option(PS_BUILD_ADAPTER "Build the opt-in DX11 and DX12 application adapters" ON)
option(PS_WITH_OPENCV "Build OpenCV camera intrinsics solver" OFF)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(json)
add_library(ps_core src/core/model.cpp src/core/geometry.cpp src/core/project.cpp src/core/calibration.cpp src/core/simulator.cpp src/core/color.cpp src/core/qualification.cpp src/core/scan.cpp)
target_include_directories(ps_core PUBLIC include)
target_link_libraries(ps_core PUBLIC nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(ps_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(ps_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PS_WITH_OPENCV)
  find_package(OpenCV 4 REQUIRED COMPONENTS core calib3d imgproc imgcodecs videoio)
  target_sources(ps_core PRIVATE src/core/opencv_calibration.cpp)
  target_link_libraries(ps_core PUBLIC ${OpenCV_LIBS})
  target_include_directories(ps_core PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_compile_definitions(ps_core PUBLIC PS_WITH_OPENCV=1)
endif()
add_executable(projection-cli src/cli/main.cpp)
target_link_libraries(projection-cli PRIVATE ps_core)
if(WIN32)
  add_library(ps_d3d11 src/windows/d3d11_renderer.cpp src/windows/render_assets.cpp src/windows/capture_transform.cpp src/windows/desktop_capture.cpp src/windows/window_capture.cpp src/windows/displays.cpp)
  target_include_directories(ps_d3d11 PUBLIC include)
  target_compile_definitions(ps_d3d11 PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(ps_d3d11 PUBLIC ps_core d3d11 dxgi d3dcompiler dwmapi windowscodecs ole32 shcore runtimeobject)
  add_library(ps_d3d12 src/windows/d3d12_renderer.cpp)
  target_link_libraries(ps_d3d12 PUBLIC ps_d3d11 d3d12)
  add_library(projection-render-api12 SHARED src/windows/renderer12_api.cpp)
  set_target_properties(projection-render-api12 PROPERTIES PREFIX "")
  target_link_libraries(projection-render-api12 PRIVATE ps_d3d12)
  add_executable(projection-renderer WIN32 src/windows/worker.cpp)
  add_executable(projection-benchmark src/windows/benchmark.cpp)
  target_link_libraries(projection-benchmark PRIVATE ps_d3d11)
  add_library(projection-render-api SHARED src/windows/renderer_api.cpp)
  set_target_properties(projection-render-api PROPERTIES PREFIX "")
  target_link_libraries(projection-render-api PRIVATE ps_d3d11)
  target_link_libraries(projection-renderer PRIVATE ps_d3d11 shell32)
  if(MINGW)
    target_link_options(projection-renderer PRIVATE -municode)
  endif()
  if(PS_BUILD_ADAPTER)
    FetchContent_Declare(minhook URL https://github.com/TsudaKageyu/minhook/archive/refs/tags/v1.3.4.tar.gz URL_HASH SHA256=1aebeae4ca898330c507860acc2fca2eb335fe446a3a2b8444c3bf8b2660a14e DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(minhook)
    add_library(projection-dx11 SHARED src/windows/dx11_adapter.cpp)
    set_target_properties(projection-dx11 PROPERTIES PREFIX "")
    target_link_libraries(projection-dx11 PRIVATE ps_d3d11 minhook)
    add_library(projection-dx12 SHARED src/windows/dx12_adapter.cpp)
    set_target_properties(projection-dx12 PROPERTIES PREFIX "")
    target_link_libraries(projection-dx12 PRIVATE ps_d3d12 minhook)
    add_executable(projection-attach src/windows/attach.cpp)
    target_compile_definitions(projection-attach PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_executable(projection-dx11-sample WIN32 tests/dx11_sample.cpp)
    target_link_libraries(projection-dx11-sample PRIVATE ps_d3d11)
    if(MINGW)
      target_link_options(projection-attach PRIVATE -municode)
      target_link_options(projection-dx11-sample PRIVATE -municode)
      target_link_options(projection-dx11 PRIVATE -static -static-libgcc -static-libstdc++)
      target_link_options(projection-dx12 PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    target_link_libraries(projection-dx11-sample PRIVATE shell32)
  endif()
endif()
if(PS_BUILD_EDITOR)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network WebSockets)
  qt_standard_project_setup()
  qt_add_executable(projection-studio src/editor/main.cpp src/editor/main_window.cpp src/editor/mesh_view.cpp src/editor/control_server.cpp src/editor/preview_service.cpp src/editor/workbench.cpp src/editor/ui_self_test.cpp src/editor/setup_check_dialog.cpp src/editor/calibration_wizard.cpp)
  target_link_libraries(projection-studio PRIVATE ps_core Qt6::Widgets Qt6::Network Qt6::WebSockets)
  target_include_directories(projection-studio PRIVATE src/editor)
  if(WIN32)
    add_executable(projection-scan src/windows/scan.cpp)
    target_link_libraries(projection-scan PRIVATE ps_core Qt6::Widgets)
    target_compile_definitions(projection-scan PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(PS_WITH_OPENCV)
      target_include_directories(projection-scan PRIVATE ${OpenCV_INCLUDE_DIRS})
    endif()
    add_dependencies(projection-studio projection-scan)
    install(TARGETS projection-scan RUNTIME DESTINATION .)
    add_library(ps_setup src/windows/setup_check.cpp)
    target_link_libraries(ps_setup PUBLIC ps_d3d11 d3d12 version advapi32)
    if(MSVC)
      target_compile_options(ps_setup PRIVATE /utf-8)
    endif()
    add_executable(projection-qualify src/windows/qualify.cpp)
    target_link_libraries(projection-qualify PRIVATE ps_setup Qt6::Network)
    add_dependencies(projection-qualify projection-renderer)
    add_dependencies(projection-studio projection-qualify)
    install(TARGETS projection-qualify RUNTIME DESTINATION .)
    target_link_libraries(projection-studio PRIVATE ps_d3d11)
    set_target_properties(projection-studio PROPERTIES WIN32_EXECUTABLE TRUE)
  endif()
  install(TARGETS projection-studio RUNTIME DESTINATION .)
endif()
if(PS_BUILD_TESTS)
  enable_testing()
  add_executable(ps-tests tests/core_tests.cpp)
  target_link_libraries(ps-tests PRIVATE ps_core)
  add_test(NAME core COMMAND ps-tests)
  add_executable(ps-qualification-tests tests/qualification_tests.cpp)
  target_link_libraries(ps-qualification-tests PRIVATE ps_core)
  add_test(NAME qualification COMMAND ps-qualification-tests)
  add_executable(ps-scan-tests tests/scan_tests.cpp)
  target_link_libraries(ps-scan-tests PRIVATE ps_core)
  add_test(NAME scan COMMAND ps-scan-tests)
  if(PS_WITH_OPENCV)
    add_executable(ps-camera-tests tests/opencv_tests.cpp)
    target_link_libraries(ps-camera-tests PRIVATE ps_core)
    target_include_directories(ps-camera-tests PRIVATE ${OpenCV_INCLUDE_DIRS})
    add_test(NAME camera COMMAND ps-camera-tests)
  endif()
  if(PS_BUILD_EDITOR)
    add_executable(ps-control-tests tests/control_tests.cpp src/editor/control_server.cpp)
    target_include_directories(ps-control-tests PRIVATE src/editor)
    target_link_libraries(ps-control-tests PRIVATE ps_core Qt6::Network Qt6::WebSockets)
    add_test(NAME control COMMAND ps-control-tests)
    if(WIN32)
      add_executable(ps-scan-runner-tests tests/scan_runner_tests.cpp)
      target_link_libraries(ps-scan-runner-tests PRIVATE ps_core Qt6::Core)
      add_dependencies(ps-scan-runner-tests projection-scan)
      add_test(NAME scan-runner COMMAND ps-scan-runner-tests)
      set_tests_properties(scan-runner PROPERTIES TIMEOUT 60)
      add_executable(ps-worker-tests tests/worker_tests.cpp)
      target_link_libraries(ps-worker-tests PRIVATE ps_core Qt6::Network)
      add_dependencies(ps-worker-tests projection-renderer projection-qualify)
      add_test(NAME worker COMMAND ps-worker-tests)
      set_tests_properties(worker PROPERTIES LABELS "desktop" TIMEOUT 90)
    endif()
  endif()
  if(WIN32)
    add_executable(ps-gpu-tests tests/gpu_tests.cpp)
    target_link_libraries(ps-gpu-tests PRIVATE ps_d3d11 projection-render-api)
    add_test(NAME gpu COMMAND ps-gpu-tests)
    add_executable(ps-dx12-tests tests/dx12_tests.cpp)
    target_link_libraries(ps-dx12-tests PRIVATE ps_d3d12 projection-render-api12)
    add_test(NAME dx12 COMMAND ps-dx12-tests)
    if(PS_BUILD_ADAPTER)
      target_compile_definitions(projection-dx11 PRIVATE PS_ADAPTER_DIAGNOSTICS)
      target_compile_definitions(projection-dx12 PRIVATE PS_ADAPTER_DIAGNOSTICS)
      add_executable(ps-integration-tests tests/windows_integration.cpp)
      target_link_libraries(ps-integration-tests PRIVATE ps_d3d11)
      add_executable(ps-adapter-tests tests/adapter_tests.cpp)
      target_link_libraries(ps-adapter-tests PRIVATE ps_d3d12)
      add_dependencies(ps-adapter-tests projection-dx11 projection-dx12)
      add_test(NAME adapter11 COMMAND ps-adapter-tests --dx11)
      add_test(NAME adapter12 COMMAND ps-adapter-tests --dx12)
      set_tests_properties(adapter11 adapter12 PROPERTIES LABELS "desktop" TIMEOUT 60)
      # Requires an interactive Windows desktop; run explicitly, not on headless CI.
    endif()
  endif()
endif()
install(TARGETS projection-cli RUNTIME DESTINATION .)
if(WIN32)
  install(TARGETS projection-renderer projection-benchmark projection-render-api projection-render-api12 RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION sdk)
  install(FILES include/ps/plugin.h include/ps/plugin12.h DESTINATION sdk/include/ps)
  if(PS_BUILD_ADAPTER)
    install(TARGETS projection-dx11 projection-dx12 projection-attach RUNTIME DESTINATION . LIBRARY DESTINATION .)
  endif()
endif()
install(DIRECTORY examples/ DESTINATION examples)
install(FILES LICENSE COPYRIGHT README.md THIRD_PARTY_NOTICES.md DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME ProjectionStudio)
include(CPack)
